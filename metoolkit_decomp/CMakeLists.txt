# metoolkit_decomp — the recovered Karma, as MathEngine's own library shape.
#
#   FetchContent_MakeAvailable(metoolkit_karma_decomp)   # or add_subdirectory
#   target_link_libraries(YourEngine PRIVATE metoolkit_decomp::all)
#
# That one line is the whole integration: it brings the recovered implementation,
# metoolkit's public headers, and ME_FORCE_CDECL, which is not optional (see below).
#
# ---------------------------------------------------------------------------
# WHY THERE IS ONE ARCHIVE AND SIXTEEN OBJECT LIBRARIES, RATHER THAN SIXTEEN
# ARCHIVES. THIS IS THE LOAD-BEARING DECISION IN THIS FILE.
#
# MathEngine split Karma into sixteen static libraries with CIRCULAR dependencies
# between them — McdCommon <-> McdPrimitives, Mst -> McdSpace/Mdt — which is why
# linking the shipped originals requires `-Wl,--start-group ... --end-group` to
# make ld re-scan until the symbol set closes. The recovered objects inherit
# every one of those cycles, because they are the same code.
#
# So the per-archive grouping here is an OBJECT library each: it reproduces the
# 2003 layout faithfully, it is what `library-map.txt` records, and it costs
# nothing at link time. `metoolkit_decomp::all` then collects every object into
# ONE archive, where the linker sees the whole symbol set at once and the cycles
# cannot bite. That is exactly how the engine has linked it since 2026-08-27, and
# it is why its own CMake carries the note "NO --start-group HERE, and that is
# not an oversight".
#
# If you genuinely want sixteen separate `.a` files — to reproduce a 2003 link
# line rather than to link a program — set METOOLKIT_DECOMP_SPLIT_ARCHIVES=ON and
# read the warning it prints. You will need the group.
# ---------------------------------------------------------------------------

cmake_minimum_required(VERSION 3.16)
project(metoolkit_decomp LANGUAGES C)

set(MD_ROOT "${CMAKE_CURRENT_LIST_DIR}")
set(MD_REPO "${CMAKE_CURRENT_LIST_DIR}/..")

# The Karma SDK's public headers. They are portable C and were never the thing
# that was missing — what was missing is an implementation off 32-bit x86.
set(METOOLKIT_DIR "${MD_REPO}/metoolkit" CACHE PATH
    "Karma SDK root: include/ + lib.{chk,rel}/")
if(NOT EXISTS "${METOOLKIT_DIR}/include/Mcd.h")
  message(FATAL_ERROR
    "metoolkit_decomp: no Karma headers at ${METOOLKIT_DIR}/include. "
    "They ship in this repository; set METOOLKIT_DIR to override.")
endif()

# Karma's public headers are split per library and several include a sibling by
# bare name, so all nine directories have to be on the path or the failure is a
# missing type three headers deep.
set(MD_METOOLKIT_INCLUDES
    "${METOOLKIT_DIR}/include"
    "${METOOLKIT_DIR}/include/MdtBcl"
    "${METOOLKIT_DIR}/include/MdtKea"
    "${METOOLKIT_DIR}/include/McdCommon"
    "${METOOLKIT_DIR}/include/McdPrimitives"
    "${METOOLKIT_DIR}/include/McdFrame"
    "${METOOLKIT_DIR}/include/Mst"
    "${METOOLKIT_DIR}/include/MeGlobals"
    "${METOOLKIT_DIR}/include/MeApp")

# ---------------------------------------------------------------------------
# Read library-map.txt. It is generated by decomp/tools/split_libraries.py from
# `ar t` over the shipped archives — the split is read off the SDK, never
# guessed — and src/ is laid out to match it.
# ---------------------------------------------------------------------------
file(STRINGS "${MD_ROOT}/library-map.txt" MD_MAP_LINES REGEX "^[^#]")
set(MD_LIBRARIES "")
foreach(line IN LISTS MD_MAP_LINES)
  string(REGEX MATCH "^([A-Za-z0-9_]+)[ \t]+(.+)$" _m "${line}")
  if(NOT _m)
    message(FATAL_ERROR "metoolkit_decomp: unparsable library-map.txt line: ${line}")
  endif()
  set(_lib "${CMAKE_MATCH_1}")
  set(_src "${MD_ROOT}/src/${_lib}/${CMAKE_MATCH_2}")
  if(NOT EXISTS "${_src}")
    message(FATAL_ERROR
      "metoolkit_decomp: library-map.txt names ${_src}, which does not exist. "
      "Regenerate both with decomp/tools/split_libraries.py.")
  endif()
  list(APPEND MD_SRC_${_lib} "${_src}")
  list(APPEND MD_LIBRARIES "${_lib}")
endforeach()
list(REMOVE_DUPLICATES MD_LIBRARIES)
list(LENGTH MD_MAP_LINES MD_TOTAL_SOURCES)

# ---------------------------------------------------------------------------
# THE FLAGS ARE NOT NEGOTIABLE, and they are the same ones
# decomp/test/standalone/wasm_check.sh validates 145/145 against.
#
# Decompiled C puns pointers through integer slots constantly, so
# -Wno-int-conversion / -Wno-incompatible-pointer-types are LOAD-BEARING rather
# than sloppy. They used to be safe only because every target had 32-bit
# pointers; since the LP64 post-passes landed (2026-08-29) the sources are
# correct at both widths and decomp/test/standalone/lp64_pipeline.sh is the
# gate that says so. They still hide nothing you want to see: the compiler has
# no opinion worth hearing about generated code of this shape.
#
# -fno-strict-aliasing is not optional either — the recovery reinterprets
# storage the way the original machine code did, and a strict-aliasing build
# miscompiles it. That was diagnosed once as a crash in the code and was the
# compiler; see decomp/proven.txt.
# ---------------------------------------------------------------------------
set(MD_C_FLAGS -std=gnu99 -fno-strict-aliasing -w
               -Wno-int-conversion -Wno-incompatible-pointer-types)

option(METOOLKIT_DECOMP_CENSUS
       "Instrument every recovered function with a call counter (NEVER a shipping build)" OFF)
option(METOOLKIT_DECOMP_SPLIT_ARCHIVES
       "Emit sixteen separate archives instead of one (you will need --start-group)" OFF)

# ---------------------------------------------------------------------------
# One OBJECT library per shipped archive.
# ---------------------------------------------------------------------------
foreach(lib IN LISTS MD_LIBRARIES)
  add_library(metoolkit_decomp_${lib} OBJECT ${MD_SRC_${lib}})
  add_library(metoolkit_decomp::${lib} ALIAS metoolkit_decomp_${lib})
  target_include_directories(metoolkit_decomp_${lib} PUBLIC
      "${MD_ROOT}/include" ${MD_METOOLKIT_INCLUDES})
  target_compile_options(metoolkit_decomp_${lib} PRIVATE ${MD_C_FLAGS})
  # ★ ME_FORCE_CDECL IS PUBLIC ON THE LIBRARY'S OWN OBJECTS, NOT ONLY INTERFACE.
  # See the long note on the aggregate below for what it does. What it does HERE
  # is stop the library miscompiling ITSELF: on Windows MinGW predefines WIN32,
  # so metoolkit's MeCall.h declares the API __stdcall, and kd_types.h's plain
  # cdecl typedefs then collide with it —
  #     kd_types.h:136: error: conflicting types for 'CDataFreeFunc'
  # which is precisely the diagnostic MeCall.h's comment says is "the only
  # warning anything gives". Relying on the consumer to have defined it before
  # add_subdirectory() ran is a build-order dependency, and it failed exactly
  # that way the first time engine-ut2004 built Windows against this package.
  target_compile_definitions(metoolkit_decomp_${lib} PUBLIC ME_FORCE_CDECL=1)
  target_compile_definitions(metoolkit_decomp_${lib} PRIVATE LINUX=1)
  set_property(TARGET metoolkit_decomp_${lib} PROPERTY POSITION_INDEPENDENT_CODE ON)
  if(METOOLKIT_DECOMP_CENSUS)
    target_compile_options(metoolkit_decomp_${lib} PRIVATE -finstrument-functions)
  endif()
endforeach()

# ---------------------------------------------------------------------------
# The aggregate: one archive, every object, no link order to get wrong.
# ---------------------------------------------------------------------------
set(MD_ALL_OBJECTS "")
foreach(lib IN LISTS MD_LIBRARIES)
  list(APPEND MD_ALL_OBJECTS $<TARGET_OBJECTS:metoolkit_decomp_${lib}>)
endforeach()

add_library(metoolkit_decomp STATIC ${MD_ALL_OBJECTS})
add_library(metoolkit_decomp::all ALIAS metoolkit_decomp)
target_include_directories(metoolkit_decomp PUBLIC
    "${MD_ROOT}/include" ${MD_METOOLKIT_INCLUDES})

# ★ ME_FORCE_CDECL IS INTERFACE, AND THAT IS THE POINT OF PUTTING IT HERE.
#
# MathEngine's Windows build declared the whole API __stdcall, and MinGW-w64
# predefines WIN32 itself, so a Windows consumer would get __stdcall
# declarations for free. The recovered sources were decompiled from the Linux
# i386 build and are CDECL, every one of them. A consumer that misses this
# define calls them stdcall while they return cdecl — caller and callee
# disagreeing about who pops the arguments, which is stack corruption on every
# Karma call with nothing to diagnose it.
#
# It used to be the consumer's job to remember. It is now impossible to forget:
# link the target, get the define. metoolkit/include/MeCall.h has the guard.
#
# ⚠ Making kd_types.h's typedefs say MEAPI too would silence the compiler and
# KEEP the bug. The conflicting-types error this resolves is the only warning
# anything gives; the fix has to change the CONVENTION, not the declaration
# that reports it.
target_compile_definitions(metoolkit_decomp INTERFACE ME_FORCE_CDECL=1)

if(METOOLKIT_DECOMP_CENSUS)
  # kd_instr.c is a separate TU compiled WITHOUT -finstrument-functions, or the
  # counter recurses into itself.
  #
  # ⚠ -no-pie IS LOAD-BEARING AND GOES ON THE EXECUTABLE, not here. kd_instr.c
  # records RUNTIME addresses; `nm` prints link-time offsets, and in a PIE those
  # differ by the load bias, so every address resolves to nothing and the report
  # reads "0 of 2025 functions entered" — indistinguishable from "the physics
  # never ran". Putting it on this library does NOTHING, which is how it was got
  # wrong the first time.
  add_library(metoolkit_decomp_instr STATIC
      "${MD_REPO}/decomp/test/standalone/kd_instr.c")
  target_compile_options(metoolkit_decomp_instr PRIVATE -std=gnu99 -w)
  set_property(TARGET metoolkit_decomp_instr PROPERTY POSITION_INDEPENDENT_CODE ON)
  target_link_libraries(metoolkit_decomp PUBLIC metoolkit_decomp_instr)
  message(STATUS
    "metoolkit_decomp: CENSUS build — instrumented, NOT a shipping build. "
    "Link the EXECUTABLE with -no-pie or the report reads zero.")
endif()

if(METOOLKIT_DECOMP_SPLIT_ARCHIVES)
  message(WARNING
    "metoolkit_decomp: SPLIT_ARCHIVES is on. The sixteen libraries have circular "
    "dependencies (McdCommon <-> McdPrimitives, Mst -> McdSpace/Mdt), so a plain "
    "ordered link leaves undefined references — wrap them in -Wl,--start-group "
    "... -Wl,--end-group. metoolkit_decomp::all exists to make that unnecessary.")
  foreach(lib IN LISTS MD_LIBRARIES)
    add_library(metoolkit_decomp_${lib}_a STATIC $<TARGET_OBJECTS:metoolkit_decomp_${lib}>)
    set_target_properties(metoolkit_decomp_${lib}_a PROPERTIES OUTPUT_NAME "${lib}")
  endforeach()
endif()

list(LENGTH MD_LIBRARIES MD_LIB_COUNT)
message(STATUS
  "metoolkit_decomp: ${MD_TOTAL_SOURCES} source(s) across ${MD_LIB_COUNT} librar(ies) "
  "-> metoolkit_decomp::all")
